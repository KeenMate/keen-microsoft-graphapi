EXAMPLES
--------

    # Get user's drive
    {:ok, drive} = GraphApi.Files.get_drive("user-id")

    # List root folder contents
    {:ok, %{"value" => items}} = GraphApi.Files.list_root_children(drive["id"])

    # Get item by path
    {:ok, item} = GraphApi.Files.get_item_by_path("drive-id", "Documents/report.docx")

    # Download file
    {:ok, binary_content} = GraphApi.Files.download_content("drive-id", "item-id")

    # Upload small file (< 4 MB)
    {:ok, item} = GraphApi.Files.upload_small("drive-id", "path/file.txt", file_content)

    # Create upload session for large file
    {:ok, session} = GraphApi.Files.create_upload_session("drive-id", "path/large.zip")
    # session["uploadUrl"] => URL for chunk uploads

    # Create folder
    {:ok, folder} = GraphApi.Files.create_folder("drive-id", "parent-id", %{"name" => "New Folder"})

    # Search
    {:ok, %{"value" => results}} = GraphApi.Files.search("drive-id", "quarterly report")

    # Create sharing link
    {:ok, link} = GraphApi.Files.create_sharing_link("drive-id", "item-id", %{
      "type" => "view",
      "scope" => "organization"
    })
