EXAMPLES
--------

    # List users with OData query
    alias GraphApi.OData
    query = OData.new() |> OData.select(["displayName", "mail"]) |> OData.top(10)
    {:ok, response} = GraphApi.Users.list(query: query)

    # Get user with schema casting
    {:ok, user} = GraphApi.Users.get("user@contoso.com", as: GraphApi.Schema.User)

    # Multi-tenant
    client = GraphApi.Client.new(config: config)
    {:ok, users} = GraphApi.Users.list(client: client)

    # Delegated
    {:ok, me} = GraphApi.Users.get("me", access_token: user_token)

    # Batch variant
    request = GraphApi.Users.list_query(query: query, as: GraphApi.Schema.User)
    # => %Batch.Request{method: "GET", url: "/users", ...}
