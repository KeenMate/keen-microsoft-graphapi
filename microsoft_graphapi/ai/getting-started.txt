GETTING STARTED
===============

Package: keen_microsoft_graphapi
Module: GraphApi


INSTALLATION
------------

Add to mix.exs:

    def deps do
      [
        {:keen_microsoft_graphapi, "~> 1.0.0-rc.1"}
      ]
    end

Run:

    mix deps.get


SINGLE-TENANT CONFIGURATION
----------------------------

Add to config/runtime.exs:

    config :keen_microsoft_graphapi, :config,
      tenant_id: System.fetch_env!("AZURE_TENANT_ID"),
      client_id: System.fetch_env!("AZURE_CLIENT_ID"),
      client_secret: System.fetch_env!("AZURE_CLIENT_SECRET")

With this configuration, resource modules can be called directly
without passing a client. The library automatically acquires and
caches OAuth tokens via the TokenStore GenServer.

    {:ok, %{"value" => users}} = GraphApi.Users.list()
    {:ok, user} = GraphApi.Users.get("user@contoso.com")


MULTI-TENANT CONFIGURATION
---------------------------

Build an explicit config and client per tenant:

    config = GraphApi.Config.new!(
      tenant_id: "tenant-aaa",
      client_id: "client-bbb",
      client_secret: "secret-ccc"
    )

    client = GraphApi.Client.new(config: config)
    {:ok, users} = GraphApi.Users.list(client: client)

Each config/client pair manages its own tokens independently.


FIRST API CALL
--------------

After configuring, make your first call:

    # List all users (returns raw JSON maps)
    {:ok, %{"value" => users}} = GraphApi.Users.list()

    # Get a specific user by ID or userPrincipalName
    {:ok, user} = GraphApi.Users.get("user@contoso.com")

    # With OData query parameters
    alias GraphApi.OData

    query = OData.new()
      |> OData.select(["displayName", "mail", "id"])
      |> OData.top(10)

    {:ok, response} = GraphApi.Users.list(query: query)

    # With schema casting (JSON -> typed Elixir structs)
    {:ok, user} = GraphApi.Users.get("user-id", as: GraphApi.Schema.User)
    # => %GraphApi.Schema.User{id: "abc", display_name: "Alice", ...}


BETA ENDPOINT
-------------

To use the Microsoft Graph beta API:

    # Via client option
    client = GraphApi.Client.new(api_version: :beta)
    {:ok, users} = GraphApi.Users.list(client: client)

    # Via config
    config = GraphApi.Config.new!(
      tenant_id: "...",
      client_id: "...",
      client_secret: "...",
      api_version: :beta
    )
    client = GraphApi.Client.new(config: config)

Beta base URL: https://graph.microsoft.com/beta
v1.0 base URL: https://graph.microsoft.com/v1.0 (default)


COMMON OPTIONS FOR ALL RESOURCE FUNCTIONS
-----------------------------------------

Every resource function (e.g., Users.list, Groups.get) accepts:

  :client          - A %Req.Request{} from GraphApi.Client.new()
  :query           - An %OData{} struct or map of query params
  :as              - Schema/View module to cast response into
  :access_token    - Per-request OAuth token (delegated permissions)
  :client_request_id - Set to true (auto UUID) or a string for request correlation
