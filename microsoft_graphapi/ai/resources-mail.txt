  GraphApi.Mail.send_mail(user_id, message, opts \\ [])
  POST /users/{user_id}/sendMail
  @spec send_mail(String.t(), map(), keyword()) :: :ok | {:error, term()}
  Sends a mail on behalf of a user.
  The message map should include :subject, :body, and :toRecipients.
  Optional :save_to_sent_items keyword in opts (default: true).
  Body sent: {"message": {...}, "saveToSentItems": true}
  Note: atom keys in the message map are automatically stringified.

EXAMPLES
--------

    # List messages
    {:ok, %{"value" => messages}} = GraphApi.Mail.list_messages("user-id")

    # Send mail (atom keys are auto-stringified)
    :ok = GraphApi.Mail.send_mail("user-id", %{
      subject: "Hello",
      body: %{contentType: "Text", content: "Hi there"},
      toRecipients: [%{emailAddress: %{address: "bob@contoso.com"}}]
    })

    # Send mail with string keys
    :ok = GraphApi.Mail.send_mail("user-id", %{
      "subject" => "Hello",
      "body" => %{"contentType" => "Text", "content" => "Hi there"},
      "toRecipients" => [%{"emailAddress" => %{"address" => "bob@contoso.com"}}]
    })

    # Create draft
    {:ok, draft} = GraphApi.Mail.create_draft("user-id", %{
      subject: "Draft Subject",
      body: %{contentType: "Text", content: "Draft body"}
    })

    # List folders
    {:ok, %{"value" => folders}} = GraphApi.Mail.list_mail_folders("user-id")

    # Track message changes
    {:ok, page} = GraphApi.Mail.messages_delta("user-id")
